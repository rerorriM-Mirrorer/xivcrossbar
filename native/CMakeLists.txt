cmake_minimum_required(VERSION 3.20)
project(xivcrossbar_gamepad LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_executable(gamepad_fixture fixture.cpp)
target_compile_features(gamepad_fixture PRIVATE cxx_std_17)
add_test(NAME gamepad_fixture COMMAND gamepad_fixture)

if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
        message(FATAL_ERROR "Configure with -A Win32")
    endif()
    include(FetchContent)
    FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
        GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
    FetchContent_MakeAvailable(minhook)
    add_library(xb_gamepad SHARED bridge.cpp)
    target_compile_features(xb_gamepad PRIVATE cxx_std_17)
    target_link_libraries(xb_gamepad PRIVATE minhook dxguid)
    add_executable(gamepad_load_fixture load_fixture.cpp)
    add_dependencies(gamepad_load_fixture xb_gamepad)
    add_test(NAME gamepad_load_fixture COMMAND gamepad_load_fixture)
endif()
