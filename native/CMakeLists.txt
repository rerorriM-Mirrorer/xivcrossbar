set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_executable(gamepad_fixture fixture.cpp)
target_compile_features(gamepad_fixture PRIVATE cxx_std_17)
add_test(NAME gamepad_fixture COMMAND gamepad_fixture)

if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
        message(FATAL_ERROR "Configure with -A Win32")
    endif()
    include(FetchContent)
    FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
        GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
    FetchContent_MakeAvailable(minhook)
    add_library(xb_gamepad SHARED bridge.cpp lua_module.cpp)
    set_target_properties(xb_gamepad PROPERTIES OUTPUT_NAME xb_gamepad_messages)
    target_compile_features(xb_gamepad PRIVATE cxx_std_17)
    target_link_libraries(xb_gamepad PRIVATE minhook dxguid user32 winmm)
    add_library(xb_controller_logic SHARED controller_logic.cpp controller_actions.cpp)
    set_target_properties(xb_controller_logic PROPERTIES OUTPUT_NAME xb_controller_logic_1)
    target_compile_features(xb_controller_logic PRIVATE cxx_std_17)
    add_custom_command(TARGET xb_controller_logic POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:xb_controller_logic>"
            "$<TARGET_FILE_DIR:xb_controller_logic>/xb_controller_logic_2.dll")
    add_library(xb_shoulder_logic SHARED controller_logic.cpp controller_actions.cpp)
    set_target_properties(xb_shoulder_logic PROPERTIES OUTPUT_NAME xb_controller_logic_3)
    target_compile_features(xb_shoulder_logic PRIVATE cxx_std_17)
    target_compile_definitions(xb_shoulder_logic PRIVATE XB_SHOULDER_GESTURES)

    add_executable(gamepad_load_fixture load_fixture.cpp)
    add_dependencies(gamepad_load_fixture xb_gamepad xb_controller_logic xb_shoulder_logic lua_host_fixture)
    target_compile_features(gamepad_load_fixture PRIVATE cxx_std_17)
    add_test(NAME gamepad_load_fixture COMMAND gamepad_load_fixture)
    set_tests_properties(gamepad_load_fixture PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:gamepad_load_fixture>")
endif()
